class Solution(object):
    def containsDuplicate(self, nums):
        """
        :type nums: List[int]
        :rtype: bool
        """
        seen=set()
        for num in nums:
            if num in seen:
                return True;
                break;
            seen.add(num);
        else:
            return False;
