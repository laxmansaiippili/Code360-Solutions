class Solution(object):
    def moveZeroes(self, nums):
        """
        :type nums: List[int]
        :rtype: None Do not return anything, 
        modify nums in-place instead.
        """
        i=1;j=0;
        n=len(nums);
        while(j<n):
            if(nums[j]==0):
                nums[j]=nums[i]
                nums[i]=0;
                i=i+1;
            else:
                j=j+1;
