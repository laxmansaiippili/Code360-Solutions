class Solution {
    public void merge(int[] nums1, int m, int[] nums2, int n) {
        int mer[]=new int[m+n];
        int i=0,j=0,k=0;
        for(;k<m+n-i-j;)
        {
            if(nums1[i]<nums2[j])
                mer[k++]=nums1[i++];
            if(nums1[i]>nums2[j])
                mer[k++]=nums2[j++];
        }
    }
}
